import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        int[] arr = new int[n];

        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }

        int k = sc.nextInt();

        if (k <= 0 || k > n) {
            System.out.println(0);
            sc.close();
            return;
        }

        Map<Integer, Integer> map = new HashMap<>();

        for (int i = 0; i < k; i++) {
            map.put(arr[i], map.getOrDefault(arr[i], 0) + 1);
        }

        int max = map.size();

        for (int i = k; i < n; i++) {
            int remove = arr[i - k];

            map.put(remove, map.get(remove) - 1);

            if (map.get(remove) == 0) {
                map.remove(remove);
            }

            map.put(arr[i], map.getOrDefault(arr[i], 0) + 1);

            max = Math.max(max, map.size());
        }

        System.out.println(max);

        sc.close();
    }
}
