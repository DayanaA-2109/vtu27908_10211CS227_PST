import java.util.*;

public class Main {

    public static int migratoryBirds(List<Integer> arr) {
        int[] count = new int[6];

        for (int bird : arr) {
            count[bird]++;
        }

        int maxCount = 0;
        int birdType = 0;

        for (int i = 1; i <= 5; i++) {
            if (count[i] > maxCount) {
                maxCount = count[i];
                birdType = i;
            }
        }

        return birdType;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        List<Integer> arr = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            arr.add(sc.nextInt());
        }

        System.out.println(migratoryBirds(arr));
        sc.close();
    }
}
