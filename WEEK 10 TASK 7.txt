import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        TreeSet<Integer> marks = new TreeSet<>();

        for (int i = 0; i < n; i++) {
            marks.add(sc.nextInt());
        }

        System.out.println("Unique Marks in Ascending Order:");
        for (int mark : marks) {
            System.out.println(mark);
        }

        sc.close();
    }
}
