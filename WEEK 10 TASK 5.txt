import java.util.*;

class Node {
    String song;
    Node next;

    Node(String song) {
        this.song = song;
        this.next = null;
    }
}

class Playlist {
    Node head;

    public void addSong(String song) {
        Node newNode = new Node(song);

        if (head == null) {
            head = newNode;
        } else {
            Node temp = head;
            while (temp.next != null) {
                temp = temp.next;
            }
            temp.next = newNode;
        }
    }

    public void removeSong(String song) {
        if (head == null) {
            return;
        }

        if (head.song.equalsIgnoreCase(song)) {
            head = head.next;
            return;
        }

        Node temp = head;
        while (temp.next != null &&
               !temp.next.song.equalsIgnoreCase(song)) {
            temp = temp.next;
        }

        if (temp.next != null) {
            temp.next = temp.next.next;
        }
    }

    public void displayPlaylist() {
        Node temp = head;

        if (temp == null) {
            System.out.println("Playlist is empty");
            return;
        }

        while (temp != null) {
            System.out.println(temp.song);
            temp = temp.next;
        }
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        Playlist playlist = new Playlist();

        int n = sc.nextInt();
        sc.nextLine();

        for (int i = 0; i < n; i++) {
            playlist.addSong(sc.nextLine());
        }

        System.out.println("Playlist:");
        playlist.displayPlaylist();

        String remove = sc.nextLine();
        playlist.removeSong(remove);

        System.out.println("Updated Playlist:");
        playlist.displayPlaylist();

        sc.close();
    }
}
